```
while True:
    #리스트형식으로 받고
    nums = list(map(int, input().split())) 

    #0,0,0 => 총합이 0이다
    if sum(nums)==0:
        break;

    max_num = max(nums)
    nums.remove(max_num)

    if nums[0]**2 + nums[1]**2 == max_num**2:
        print("right")
    else:
        print("wrong")
```
